function formatDate(value) {
  if (!value) {
    return '—';
  }

  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return '—';
  }

  return new Intl.DateTimeFormat(
    'en-GB',
    {
      day: '2-digit',
      month: 'short',
      year: 'numeric',
    }
  ).format(date);
}

function formatDateTime(
  value
) {
  if (!value) {
    return '—';
  }

  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return '—';
  }

  return new Intl.DateTimeFormat(
    'en-GB',
    {
      day: '2-digit',
      month: 'short',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
    }
  ).format(date);
}

function getStatusClasses(
  status
) {
  const normalizedStatus =
    String(status || '')
      .trim()
      .toLowerCase();

  if (
    [
      'approved',
      'active',
      'registered',
    ].includes(
      normalizedStatus
    )
  ) {
    return 'border-emerald-200 bg-emerald-50 text-emerald-700';
  }

  if (
    [
      'submitted',
      'under review',
      'document verification',
      'registration review',
      'awaiting registration fee',
      'payment verification',
      'admin/hr & finance review',
      'director general review',
    ].includes(
      normalizedStatus
    )
  ) {
    return 'border-amber-200 bg-amber-50 text-amber-700';
  }

  if (
    [
      'rejected',
      'cancelled',
    ].includes(
      normalizedStatus
    )
  ) {
    return 'border-red-200 bg-red-50 text-red-700';
  }

  if (
    normalizedStatus ===
    'returned for revision'
  ) {
    return 'border-blue-200 bg-blue-50 text-blue-700';
  }

  return 'border-slate-200 bg-slate-50 text-slate-700';
}

function StatusBadge({
  value,
}) {
  return (
    <span
      className={`inline-flex rounded-full border px-2.5 py-1 text-xs font-semibold ${getStatusClasses(
        value
      )}`}
    >
      {value || 'Unknown'}
    </span>
  );
}

}

function ApplicationReviewModal({
  applicationId,
  onClose,

    useNGOApplication(
      applicationId,
      {
        enabled:
          Boolean(
            applicationId
          ),
      }
    );

  const startReviewMutation =
    useStartNGOApplicationReview();

  const documentVerificationMutation =
    useCompleteDocumentVerification();

  const [
    actionMessage,
    setActionMessage,
  ] = useState('');

  const [
    actionError,
    setActionError,
  ] = useState('');

  if (!applicationId) {
    return null;
  }

  const application =
    applicationQuery.data;

  const clearActionState =
    () => {
      setActionMessage('');
      setActionError('');
    };

  const handleStartReview =
    async () => {
      const confirmed =
        window.confirm(
          'Start Ministry review of this application? The application will move to Document Verification.'
        );

      if (!confirmed) {
        return;
      }

      clearActionState();

      try {
        const result =
          await startReviewMutation.mutateAsync(
            applicationId
          );

        setActionMessage(
          result.message
        );

        await applicationQuery.refetch();
      } catch (error) {
        setActionError(
          error?.response?.data
            ?.message ||
            error?.message ||
            'Unable to start application review.'
        );
      }
    };

  const handleCompleteDocumentVerification =
    async () => {
      const confirmed =
        window.confirm(
          'Confirm that the required registration documents have been verified? The application will move to Registration / Compliance Review.'
        );

      if (!confirmed) {
        return;
      }

      clearActionState();

      try {
        const result =
          await documentVerificationMutation.mutateAsync(
            applicationId
          );

        setActionMessage(
          result.message
        );

        await applicationQuery.refetch();

            error?.message ||
            'Unable to complete document verification.'

export function NGOApplications() {
  const [
    searchTerm,
    setSearchTerm,
  ] = useState('');

  const [
    status,
    setStatus,
  ] = useState('all');

  const [
    approvalStage,
    setApprovalStage,
  ] = useState('all');

  const [
    selectedApplicationId,
    setSelectedApplicationId,
  ] = useState(null);

  const applicationsQuery =
    useNGOApplications({
      limit: 100,
    });

  const applications =
    applicationsQuery.data
      ?.items || [];

  const filteredApplications =
    useMemo(() => {
      const normalizedSearch =
        searchTerm
          .trim()
          .toLowerCase();

      return applications.filter(
        (application) => {
          const matchesSearch =
            !normalizedSearch ||
            String(
              application.applicationNumber ||
                ''
            )
              .toLowerCase()
              .includes(
                normalizedSearch
              ) ||
            String(
              application.organizationName ||
                ''
            )
              .toLowerCase()
              .includes(
                normalizedSearch
              ) ||
            String(
              application.organizationType ||
                ''
            )
              .toLowerCase()
              .includes(
                normalizedSearch
              ) ||
            String(
              application
                .applicant
                ?.fullName ||
                ''
            )
              .toLowerCase()
              .includes(
                normalizedSearch
              ) ||
            String(
              application
                .applicant
                ?.email ||
                ''
            )
              .toLowerCase()
              .includes(
                normalizedSearch
              );

          const matchesStatus =
            status === 'all' ||
            application.status ===
              status;

          const matchesStage =
            approvalStage ===
              'all' ||
            application.approvalStage ===
              approvalStage;

          return (
            matchesSearch &&
            matchesStatus &&
            matchesStage
          );
        }
      );
    }, [
      applications,
      searchTerm,
      status,
      approvalStage,
    ]);

  const submittedCount =
    applications.filter(
      (application) =>
        application.status ===
        'Submitted'
    ).length;

  const underReviewCount =
    applications.filter(
      (application) =>
        application.status ===
        'Under Review'
    ).length;

  const completedCount =
    applications.filter(
      (application) =>
        application.status ===
          'Approved' ||
        application.status ===
          'Rejected'
    ).length;

  return (
    <>
      <section className="space-y-6">
        <div>
          <h2 className="text-2xl font-semibold text-slate-900">
            Registration Applications
          </h2>

          <p className="mt-1 max-w-3xl text-sm leading-6 text-slate-500">
            Review organization
            registration applications
            submitted through the
            public JAIMS registration
            portal. Approved
            organizations will later
            move into the Registered
            Organizations registry.
          </p>
        </div>

        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
          <div className="rounded-lg border border-slate-200 bg-white p-5">
            <p className="text-sm font-medium text-slate-500">
              Total Applications
            </p>

            <p className="mt-2 text-3xl font-semibold text-slate-900">
              {
                applications.length
              }
            </p>
          </div>

          <div className="rounded-lg border border-slate-200 bg-white p-5">
            <p className="text-sm font-medium text-slate-500">
              Submitted
            </p>

            <p className="mt-2 text-3xl font-semibold text-amber-700">
              {submittedCount}
            </p>
          </div>

          <div className="rounded-lg border border-slate-200 bg-white p-5">
            <p className="text-sm font-medium text-slate-500">
              Under Review
            </p>

            <p className="mt-2 text-3xl font-semibold text-blue-700">
              {
                underReviewCount
              }
            </p>
          </div>

          <div className="rounded-lg border border-slate-200 bg-white p-5">
            <p className="text-sm font-medium text-slate-500">
              Completed
              Decisions
            </p>

            <p className="mt-2 text-3xl font-semibold text-emerald-700">
              {completedCount}
            </p>
          </div>
        </div>

        <div className="overflow-hidden rounded-lg border border-slate-200 bg-white">
          <div className="border-b border-slate-200 p-4">
            <div className="flex flex-wrap gap-3">
              <div className="relative min-w-64 flex-1">
                <Search
                  size={18}
                  className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-slate-400"
                />

                <input
                  type="search"
                  value={
                    searchTerm
                  }
                  onChange={(
                    event
                  ) =>
                    setSearchTerm(
                      event.target
                        .value
                    )
                  }
                  placeholder="Search application number, organization, applicant or email"
                  className="w-full rounded border border-slate-300 bg-white py-2.5 pl-10 pr-3 text-sm outline-none transition focus:border-civic focus:ring-2 focus:ring-civic/20"
                />
              </div>

              <select
                value={status}
                onChange={(
                  event
                ) =>
                  setStatus(
                    event.target
                      .value
                  )
                }
                className="rounded border border-slate-300 bg-white px-3 py-2.5 text-sm text-slate-700 outline-none focus:border-civic"
              >
                {statusOptions.map(
                  (option) => (
                    <option
                      key={option}
                      value={option}
                    >
                      {option ===
                      'all'
                        ? 'All statuses'
                        : option}
                    </option>
                  )
                )}
              </select>

              <select
                value={
                  approvalStage
                }
                onChange={(
                  event
                ) =>
                  setApprovalStage(
                    event.target
                      .value
                  )
                }
                className="rounded border border-slate-300 bg-white px-3 py-2.5 text-sm text-slate-700 outline-none focus:border-civic"
              >
                {stageOptions.map(
                  (option) => (
                    <option
                      key={option}
                      value={option}
                    >
                      {option ===
                      'all'
                        ? 'All review stages'
                        : option}
                    </option>
                  )
                )}
              </select>

              <button
                type="button"
                onClick={() =>
                  applicationsQuery.refetch()
                }
                disabled={
                  applicationsQuery.isFetching
                }
                className="inline-flex items-center gap-2 rounded border border-slate-300 bg-white px-3 py-2.5 text-sm font-semibold text-slate-700 transition hover:bg-slate-50 disabled:opacity-60"
              >
                <RefreshCw
                  size={17}
                  className={
                    applicationsQuery.isFetching
                      ? 'animate-spin'
                      : ''
                  }
                />

                Refresh
              </button>
            </div>
          </div>

          {applicationsQuery.isLoading ? (
            <div className="flex min-h-72 items-center justify-center p-8">
              <div className="text-center">
                <RefreshCw
                  size={30}
                  className="mx-auto animate-spin text-civic"
                />

                <p className="mt-3 text-sm font-medium text-slate-600">
                  Loading registration
                  applications...
                </p>
              </div>
            </div>
          ) : applicationsQuery.isError ? (
            <div className="flex min-h-72 items-center justify-center p-8">
              <div className="max-w-md text-center">
                <FileWarning
                  size={38}
                  className="mx-auto text-red-500"
                />

                <h3 className="mt-4 text-lg font-semibold text-slate-900">
                  Unable to load
                  registration
                  applications
                </h3>

                <p className="mt-2 text-sm leading-6 text-slate-500">
                  {applicationsQuery
                    .error?.response
                    ?.data?.message ||
                    applicationsQuery
                      .error?.message ||
                    'Applications could not be retrieved from JAIMS.'}
                </p>

                <button
                  type="button"
                  onClick={() =>
                    applicationsQuery.refetch()
                  }
                  className="mt-5 inline-flex items-center gap-2 rounded bg-civic px-4 py-2.5 text-sm font-semibold text-white"
                >
                  <RefreshCw
                    size={17}
                  />

                  Try Again
                </button>
              </div>
            </div>
          ) : filteredApplications.length ===
            0 ? (
            <div className="flex min-h-72 items-center justify-center p-8">
              <div className="max-w-md text-center">
                <div className="mx-auto grid h-14 w-14 place-items-center rounded-full bg-slate-100 text-civic">
                  <FileText
                    size={28}
                  />
                </div>

                <h3 className="mt-4 text-lg font-semibold text-slate-900">
                  {applications.length ===
                  0
                    ? 'No registration applications'
                    : 'No matching applications'}
                </h3>

                <p className="mt-2 text-sm leading-6 text-slate-500">
                  {applications.length ===
                  0
                    ? 'No organization registration applications have been submitted yet.'
                    : 'Change the search or filter values and try again.'}
                </p>
              </div>
            </div>
          ) : (
            <>
              <div className="overflow-x-auto">
                <table className="min-w-full divide-y divide-slate-200">
                  <thead className="bg-slate-50">
                    <tr>
                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Application
                      </th>

                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Organization
                      </th>

                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Applicant
                      </th>

                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Submitted
                      </th>

                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Status
                      </th>

                      <th className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Review Stage
                      </th>

                      <th className="px-4 py-3 text-right text-xs font-semibold uppercase tracking-wide text-slate-500">
                        Action
                      </th>
                    </tr>
                  </thead>

                  <tbody className="divide-y divide-slate-200 bg-white">
                    {filteredApplications.map(
                      (
                        application
                      ) => (
                        <tr
                          key={
                            application.id
                          }
                          className="transition hover:bg-slate-50"
                        >
                          <td className="whitespace-nowrap px-4 py-4">
                            <p className="text-sm font-semibold text-civic">
                              {
                                application.applicationNumber
                              }
                            </p>

                            <p className="mt-1 text-xs text-slate-500">
                              {
                                application.applicationType
                              }
                            </p>
                          </td>

                          <td className="px-4 py-4">
                            <div className="flex min-w-56 items-center gap-3">
                              <div className="grid h-9 w-9 shrink-0 place-items-center rounded bg-slate-100 text-civic">
                                <Building2
                                  size={
                                    18
                                  }
                                />
                              </div>

                              <div>
                                <p className="font-semibold text-slate-900">
                                  {
                                    application.organizationName
                                  }
                                </p>

                                <p className="mt-1 text-xs text-slate-500">
                                  {
                                    application.organizationType
                                  }
                                </p>
                              </div>
                            </div>
                          </td>

                          <td className="px-4 py-4">
                            <p className="text-sm font-medium text-slate-800">
                              {application
                                .applicant
                                ?.fullName ||
                                '—'}
                            </p>

                            <p className="mt-1 text-xs text-slate-500">
                              {application
                                .applicant
                                ?.email ||
                                '—'}
                            </p>
                          </td>

                          <td className="whitespace-nowrap px-4 py-4 text-sm text-slate-600">
                            {formatDate(
                              application.submittedAt
                            )}
                          </td>

                          <td className="whitespace-nowrap px-4 py-4">
                            <StatusBadge
                              value={
                                application.status
                              }
                            />
                          </td>

                          <td className="whitespace-nowrap px-4 py-4">
                            <StatusBadge
                              value={
                                application.approvalStage
                              }
                            />
                          </td>

                          <td className="whitespace-nowrap px-4 py-4 text-right">
                            <button
                              type="button"
                              onClick={() =>
                                setSelectedApplicationId(
                                  application.id
                                )
                              }
                              className="inline-flex items-center gap-1.5 rounded border border-slate-300 bg-white px-3 py-2 text-xs font-semibold text-slate-700 transition hover:border-civic hover:text-civic"
                            >
                              <Eye
                                size={
                                  15
                                }
                              />

                              Review
                            </button>
                          </td>
                        </tr>
                      )
                    )}
                  </tbody>
                </table>
              </div>

              <div className="flex flex-wrap items-center justify-between gap-3 border-t border-slate-200 px-4 py-3">
                <p className="text-sm text-slate-500">
                  Showing{' '}
                  <span className="font-semibold text-slate-700">
                    {
                      filteredApplications.length
                    }
                  </span>{' '}
                  of{' '}
                  <span className="font-semibold text-slate-700">
                    {
                      applications.length
                    }
                  </span>{' '}
                  applications
                </p>

                {applicationsQuery.isFetching ? (
                  <p className="inline-flex items-center gap-2 text-sm text-slate-500">
                    <RefreshCw
                      size={15}
                      className="animate-spin"
                    />

                    Updating...
                  </p>
                ) : null}
              </div>
            </>
          )}
        </div>
      </section>

      <ApplicationReviewModal
        applicationId={
          selectedApplicationId
        }
        onClose={() => {
          setSelectedApplicationId(
            null
          );
        }}
      />
    </>
  );
}
